cmake_minimum_required(VERSION 3.25)
project(WinOneCaptureNativeContracts LANGUAGES CXX)
enable_testing()
add_executable(capture_pipe_frames pipe_frames_test.cpp)
target_compile_features(capture_pipe_frames PRIVATE cxx_std_20)
target_include_directories(capture_pipe_frames PRIVATE ../../modules/capture-native)
if(MSVC)
    target_compile_options(capture_pipe_frames PRIVATE /utf-8 /W4 /WX)
endif()
add_test(NAME capture_pipe_frames COMMAND capture_pipe_frames)
add_executable(capture_pipe_transport pipe_transport_test.cpp)
target_compile_features(capture_pipe_transport PRIVATE cxx_std_20)
target_include_directories(capture_pipe_transport PRIVATE ../../modules/capture-native)
target_compile_definitions(capture_pipe_transport PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(capture_pipe_transport PRIVATE ole32)
if(MSVC)
    target_compile_options(capture_pipe_transport PRIVATE /utf-8 /W4 /WX)
endif()
add_test(NAME capture_pipe_transport COMMAND capture_pipe_transport)
