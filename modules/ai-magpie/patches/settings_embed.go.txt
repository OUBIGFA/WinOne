package settings

import (
 "errors"
 "sync/atomic"
)

// Set only by the embedding entry before a request starts. A worker owns one
// snapshot; no global save or another desktop client's file is involved.
var WinOneSettings atomic.Pointer[Settings]

// Consent belongs only to this private local profile. Other Host settings
// transactions cannot silently enable recording or overwrite its switch.
func WinOneGatewayRecording() bool { return load().GatewayConversations }
func winOneSetGatewayConversations(on bool) error {
 defer lockFile()()
 s:=load();s.GatewayConversations=on
 if err:=save(s,true);err!=nil{return err}
 return WinOneEdit(func(next *Settings)error{next.GatewayConversations=on;return nil})
}

// Upstream Load is a mutable copy while Shared is read-only. Keep the same
// contract for Host snapshots so preference editors cannot mutate a reader.
func WinOneUse(s Settings) { normalized := s.normal(); WinOneSettings.Store(&normalized) }

// Catalog refresh and gateway recording changes can arrive together. Apply a field
// edit to the latest clone instead of replacing an unrelated concurrent edit.
func WinOneEdit(edit func(*Settings) error) error {
 for {
  prior:=WinOneSettings.Load();if prior==nil{return errors.New("missing WinOne settings snapshot")}
  next:=prior.clone();if err:=edit(&next);err!=nil{return err};next=next.normal()
  if WinOneSettings.CompareAndSwap(prior,&next){return nil}
 }
}

// Gateway readers receive the current immutable settings snapshot.
func WinOneLoad() (Settings, bool) {
	s := WinOneSettings.Load()
	if s == nil {
		return Settings{}, false
	}
	return *s, true
}
