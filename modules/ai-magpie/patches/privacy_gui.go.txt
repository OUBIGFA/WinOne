package gui

import (
 "context"
 "encoding/json"
 "errors"
 "time"
 "net/http"
 "github.com/yetone/magpie/internal/pii"
 "github.com/yetone/magpie/internal/settings"
)

// Only registered behind WinOne's per-process capability authentication. No
// public gateway alias, no prompt log, and no HTML surface inside Routing.
func WinOnePrivacyHandler() http.Handler {
 mux:=http.NewServeMux()
 privacyModelRoutes(mux)
 mux.HandleFunc("POST /api/privacy-model/find",func(w http.ResponseWriter,r *http.Request){
  // Browser cookies alone never authorize an inference payload.
  if r.Header.Get("Authorization")=="" {http.Error(w,"Unauthorized",401);return}
  if !settings.Load().RedactModel {http.Error(w,"Local masking model is off",409);return}
  var body struct{Texts []string `json:"texts"`;Deadline int64 `json:"deadline"`}
  decoder:=json.NewDecoder(http.MaxBytesReader(w,r.Body,48<<20));decoder.DisallowUnknownFields()
  if decoder.Decode(&body)!=nil || len(body.Texts)>20000 {http.Error(w,"Invalid masking request",400);return}
  ctx:=r.Context();if body.Deadline>0{var cancel context.CancelFunc;ctx,cancel=context.WithDeadline(ctx,time.UnixMilli(body.Deadline));defer cancel()}
  st:=settings.Shared();model:=pii.Of(pii.Choose(st.RedactModelName,st.RedactModel))
  found,err:=model.Find(ctx,body.Texts)
  var partial *pii.Partial
  if err!=nil&&!errors.As(err,&partial) {http.Error(w,"Local masking model did not answer; request held back",503);return}
  if partial!=nil{w.Header().Set("X-WinOne-Privacy-Partial","1")}
  writeJSON(w,found)
 })
 return mux
}
