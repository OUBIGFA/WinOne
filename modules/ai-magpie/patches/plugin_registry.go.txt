package plugin

import (
 "os"
 "path/filepath"
 "strings"
 "sync"
)

// Keep Windows' registered user identity for native plugin helpers. Magpie's
// XDG configuration, auth and APPDATA browser profiles remain WinOne-owned.
// The parent AI worker retains its private home for built-in account backends.
func winOnePluginEnvironment(env []string) []string {
 profile:=os.Getenv("WINONE_AI_OS_USERPROFILE")
 if !filepath.IsAbs(profile) { return env }
 result:=make([]string,0,len(env)+1)
 for _,entry:=range env {
  name,_,_:=strings.Cut(entry,"=")
  if !strings.EqualFold(name,"USERPROFILE") { result=append(result,entry) }
 }
 return append(result,"USERPROFILE="+profile)
}

// Only code and its user-selected configuration are shared by WinOne's
// workers. AuthPath continues to use each provider's private settings.Dir.
func WinOneDirectory() string {
 directory:=os.Getenv("WINONE_AI_PLUGINS_HOME")
 if !filepath.IsAbs(directory){return ""}
 return filepath.Clean(directory)
}

var winOneBundles struct {
 sync.RWMutex
 entries []Entry
}

// Verified built-ins belong to the request snapshot, never to the user's
// third-party registry. A new install path therefore needs no file migration.
func WinOneUseBundle(entry Entry) {
 winOneBundles.Lock();defer winOneBundles.Unlock()
 for i,e:=range winOneBundles.entries{if e.Spec==entry.Spec{winOneBundles.entries[i]=entry;return}}
 winOneBundles.entries=append(winOneBundles.entries,entry)
}

func winOneEffective(l List) List {
 if WinOneDirectory()==""{return l}
 winOneBundles.RLock();defer winOneBundles.RUnlock()
 l.Plugins=append(l.Plugins,clonePlugins(winOneBundles.entries)...)
 return l
}

// The manager and durable registry expose only explicitly installed code.
// Removing a third-party package cannot erase an integrated account backend.
func WinOneManaged(l List) List {
 if WinOneDirectory()==""{return l}
 winOneBundles.RLock();defer winOneBundles.RUnlock()
 kept:=l.Plugins[:0]
 for _,entry:=range l.Plugins {
  bundled:=false
  for _,b:=range winOneBundles.entries{if b.Spec==entry.Spec{bundled=true;break}}
  if !bundled{kept=append(kept,entry)}
 }
 l.Plugins=kept
 return l
}
