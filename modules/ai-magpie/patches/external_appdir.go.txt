package appdir

import (
 "errors"
 "path/filepath"
)

// WinOneExternalRoot separates Magpie-owned files from the client locations
// carried in the dedicated worker's OS environment. Called once before any
// handlers or background workers start; never used by a shared AI process.
func WinOneExternalRoot(directory string) error {
 if !filepath.IsAbs(directory){return errors.New("invalid private external-client data directory")}
 once.Do(func(){})
 portable=filepath.Clean(directory)
 return nil
}
