// The native Gateway page owns these explicit administrative actions. They
// are available only on the authenticated private UI listener, never publicly.
func winOneGatewayAdminRoutes(mux *http.ServeMux) {
 // Only gateway-owned data: never list or read an external client's files.
 historyState:=func()any{return map[string]any{"recording":settings.Load().GatewayConversations,"recorded":sessions.HasGatewayConversations(),"sessions":listedGateway(usage.GatewaySessions(time.Time{},nil))}}
 mux.HandleFunc("GET /api/winone/gateway/conversations",func(w http.ResponseWriter,r *http.Request){writeJSON(w,historyState())})
 mux.HandleFunc("POST /api/winone/gateway/conversations/recording",func(w http.ResponseWriter,r *http.Request){
  var in struct{On,Clear bool};if json.NewDecoder(http.MaxBytesReader(w,r.Body,1024)).Decode(&in)!=nil{http.Error(w,"Invalid recording choice",400);return}
  if err:=sessions.SetGatewayRecording(in.On,in.Clear);err!=nil{fail(w,err);return};writeJSON(w,historyState())
 })
 mux.HandleFunc("POST /api/winone/gateway/conversations/read",func(w http.ResponseWriter,r *http.Request){
  var in struct{Agent,ID string};if json.NewDecoder(http.MaxBytesReader(w,r.Body,16384)).Decode(&in)!=nil{http.Error(w,"Invalid gateway session",400);return}
  transcript,err:=sessions.GatewayTranscript(in.Agent,in.ID);if err!=nil{fail(w,err);return}
  var markdown strings.Builder
  if err:=sessions.WriteGatewayMarkdown(&markdown,sessions.Session{ID:in.ID,Agent:in.Agent},in.Agent,transcript);err!=nil{fail(w,err);return}
  writeJSON(w,map[string]any{"text":markdown.String(),"cut":transcript.Cut})
 })
 mux.HandleFunc("POST /api/winone/gateway/configure",func(w http.ResponseWriter,r *http.Request){
  var in access.Key;if json.NewDecoder(http.MaxBytesReader(w,r.Body,1<<20)).Decode(&in)!=nil{http.Error(w,"Invalid key policy",400);return}
  if err:=access.WinOneConfigure(in);err!=nil{fail(w,err);return}
  keys,err:=access.List();if err!=nil{fail(w,err);return};writeJSON(w,map[string]any{"keys":withLimits(keys),"accountNames":provider.AccountNames()})
 })
 mux.HandleFunc("POST /api/winone/gateway/copy",func(w http.ResponseWriter,r *http.Request){
  var in struct{ID string `json:"id"`};if json.NewDecoder(r.Body).Decode(&in)!=nil{http.Error(w,"Invalid key",400);return}
  keys,err:=access.Export();if err!=nil{fail(w,err);return}
  for _,k:=range keys {if k.ID==in.ID{writeJSON(w,map[string]string{"secret":k.Secret});return}}
  http.Error(w,"Key not found",404)
 })
 mux.HandleFunc("POST /api/winone/gateway/issue",func(w http.ResponseWriter,r *http.Request){
  var in struct{Name string `json:"name"`;TTL string `json:"ttl"`;Models []string `json:"models"`;Accounts []string `json:"accounts"`;Tokens int64 `json:"tokens"`;Cost float64 `json:"cost"`;ReadOnly bool `json:"readOnly"`}
  if json.NewDecoder(r.Body).Decode(&in)!=nil{http.Error(w,"Invalid issued key",400);return}
  ttl,err:=time.ParseDuration(in.TTL);if err!=nil{fail(w,err);return}
  key,err:=access.IssueKey(access.Issue{Name:in.Name,TTL:ttl,Models:in.Models,Accounts:in.Accounts,Tokens:in.Tokens,Cost:in.Cost,ReadOnly:in.ReadOnly});if err!=nil{fail(w,err);return}
  writeJSON(w,key)
 })
 mux.HandleFunc("GET /api/winone/gateway/oidc",func(w http.ResponseWriter,r *http.Request){rules,err:=oidc.List();if err!=nil{fail(w,err);return};writeJSON(w,map[string]any{"rules":rules})})
 mux.HandleFunc("POST /api/winone/gateway/oidc",func(w http.ResponseWriter,r *http.Request){
  var in oidc.Rule;if json.NewDecoder(r.Body).Decode(&in)!=nil{http.Error(w,"Invalid identity rule",400);return}
  rule,err:=oidc.Add(in);if err!=nil{fail(w,err);return};writeJSON(w,rule)
 })
 mux.HandleFunc("POST /api/winone/gateway/oidc/remove",func(w http.ResponseWriter,r *http.Request){
  var in struct{ID string `json:"id"`};if json.NewDecoder(r.Body).Decode(&in)!=nil{http.Error(w,"Invalid identity rule",400);return}
  if err:=oidc.Remove(in.ID);err!=nil{fail(w,err);return};writeJSON(w,map[string]bool{"removed":true})
 })
}

// Only the original read handler is exposed to the private one-shot worker.
// Archive switches remain in WinOne's saved settings transaction.
func WinOneArchiveHandler() http.Handler {
 original:=http.NewServeMux();archiveRoutes(original)
 mux:=http.NewServeMux();mux.Handle("GET /api/archive",original)
 return mux
}
func WinOneWithPagePrefix(r *http.Request,prefix string)*http.Request{return r.WithContext(context.WithValue(r.Context(),winOnePagePrefixKey{},prefix))}
func winOnePagePrefix(r *http.Request)string{prefix,_:=r.Context().Value(winOnePagePrefixKey{}).(string);return prefix}

// WinOne supplies its own navigation, provider settings and process lifetime.
// Routing and plugin operations keep their original endpoints. The isolated
// documents use WinOne presentation adapters, never the upstream app shell.
func WinOneRoutingHandler(gw *gateway.Server, bundled []string) http.Handler {
 served.Store(gw)
 mux:=http.NewServeMux()
 groupRoutes(mux);traceRoutes(mux);contextRoutesAPI(mux);maskingRoutes(mux);callerKeyRoutes(mux);winOneGatewayAdminRoutes(mux)
 // Preview uses the actual upstream pattern parser/matcher, never writes a group.
 mux.HandleFunc("POST /api/groups/preview",func(w http.ResponseWriter,r *http.Request){
  var in provider.Group;if json.NewDecoder(http.MaxBytesReader(w,r.Body,256<<10)).Decode(&in)!=nil{http.Error(w,"Invalid group preview",400);return}
  patterns,err:=provider.CleanPatterns(in.Match);if err!=nil{fail(w,err);return};in.Match=patterns
  writeJSON(w,map[string]any{"patterns":provider.PatternHits(in)})
 })
 pluginRoutes(mux,nil);pluginPageRoutes(mux,true)
 // Routing's body/HTTP-header panels may fetch an explicitly kept archive.
 // Export still goes through the native Host and its destination protection.
 archiveRead:=WinOneArchiveHandler();mux.Handle("GET /api/archive",archiveRead)
 // These read-only allowances belong to the routing account rows. Register
 // the original handlers privately without exposing the Usage application.
 quotaMux:=http.NewServeMux();usageRoutes(quotaMux,nil)
 mux.Handle("GET /api/usage/quotas",quotaMux)
 mux.Handle("GET /api/usage/quotas/history",quotaMux)
 mux.HandleFunc("GET /api/providers",func(w http.ResponseWriter,r *http.Request){writeJSON(w,providersState())})
 // Provider editor status belongs to the live gateway, not a one-shot worker.
 mux.HandleFunc("GET /api/claude/mode",func(w http.ResponseWriter,r *http.Request){mode,err:=gateway.WinOneClaudeMode();if err!=nil{fail(w,err);return};writeJSON(w,mode)})
 // The page's own settings refresh calls setLocale; keep the value in the
 // form its translation table understands instead of a regional tag.
 // The standalone page needs the same feature state as the upstream shell,
 // including the off-state explanation and its one-click enable action.
 type routingSettings struct {settings.Settings;Builtins []builtinJSON `json:"builtins"`;JSRoutings []middleware.Routing `json:"jsRoutings"`}
 winOneSettings:=func()routingSettings{s:=settings.Load();s.GitHubToken="";s.OTel.Headers=nil;if strings.HasPrefix(s.Lang,"zh"){s.Lang="zh"}else if s.Lang!="system"&&!strings.HasPrefix(s.Lang,"en"){s.Lang="en"};return routingSettings{Settings:s,Builtins:builtinsState(),JSRoutings:middleware.Routings()}}
 mux.HandleFunc("GET /api/state",func(w http.ResponseWriter,r *http.Request){writeJSON(w,map[string]any{"agents":[]any{map[string]any{"id":"winone","name":"WinOne","fields":[]any{}}},"profiles":[]any{},"settings":winOneSettings()})})
 mux.HandleFunc("GET /api/settings",func(w http.ResponseWriter,r *http.Request){writeJSON(w,winOneSettings())})
 var preferenceMu sync.Mutex
 mux.HandleFunc("POST /api/settings/plain-names",func(w http.ResponseWriter,r *http.Request){
  var body struct{Mode string};if json.NewDecoder(r.Body).Decode(&body)!=nil{http.Error(w,"Invalid model name preference",400);return}
  preferenceMu.Lock();defer preferenceMu.Unlock()
  if err:=provider.SetSuffixMode(body.Mode);err!=nil{fail(w,err);return}
  // SetSuffixMode uses the upstream validation and durable settings store.
  // Publish just its two fields into the active Host snapshot after saving.
  if err:=settings.WinOneEdit(func(s *settings.Settings)error{s.PlainNames=body.Mode==provider.SuffixOff;s.PlainOwnNames=body.Mode==provider.SuffixOwn;return nil});err!=nil{fail(w,err);return}
  writeJSON(w,winOneSettings())
 })
 mux.HandleFunc("POST /api/provider/{action}",func(w http.ResponseWriter,r *http.Request){
  var body struct{ID,Routing,Affinity string;Sink bool};if json.NewDecoder(r.Body).Decode(&body)!=nil {http.Error(w,"Invalid routing preference",400);return}
  var err error
  switch r.PathValue("action"){case "route":err=provider.SetRouting(body.ID,body.Routing);case "affinity":err=provider.SetAffinity(body.ID,body.Affinity);case "sink":err=provider.SetSink(body.ID,body.Sink);default:http.NotFound(w,r);return}
  if err!=nil {fail(w,err);return};writeJSON(w,providersState())
 })
 winOneCallRoutes(mux)
 mux.HandleFunc("GET /boot.js",func(w http.ResponseWriter,r *http.Request){w.Header().Set("Content-Type","text/javascript; charset=utf-8");lang:=settings.Load().Lang;if strings.HasPrefix(lang,"zh"){lang="zh"}else{lang="en"};b,_:=json.Marshal(map[string]any{"lang":lang,"theme":settings.Load().Theme,"web":true,"bundledPackages":bundled});w.Write(append(append([]byte("window.bootPrefs="),b...),';'))})
 files,_:=fs.Sub(assets,"assets")
 static:=http.FileServer(http.FS(files))
 mux.HandleFunc("GET /plugins.html",func(w http.ResponseWriter,r *http.Request){
  b,err:=fs.ReadFile(files,"winone-plugins.html");if err!=nil{http.Error(w,"Plugin page unavailable",500);return}
  w.Header().Set("Content-Type","text/html; charset=utf-8");w.Write(b)
 })
 for _,path:=range []string{"masking.html","winone-masking-host.js","winone-masking-prehost.js","winone-masking-client.js","winone-masking-shared.js","winone-masking-page.js","winone-masking.css","plugins/redact/i18n.js","plugins/redact/client.js","plugins/redact/redact.css","plugins.css","winone-plugin-pages.js","winone-plugin-bridge.js","winone-plugin-host.js","winone-plugin-manager.js","winone-plugin-shared.js","winone-plugins.css"}{mux.Handle("GET /"+path,static)}
 mux.HandleFunc("GET /{$}",func(w http.ResponseWriter,r *http.Request){
  b,err:=fs.ReadFile(files,"winone-routing.html");if err!=nil{http.Error(w,"Routing page unavailable",500);return}
  w.Header().Set("Content-Type","text/html; charset=utf-8");w.Write(b)
 })
 // Other Magpie applications cannot be opened by a deep link inside Routing.
 for _,path:=range []string{"app.css","context.css","compat.js","theme.js","kernel.js","i18n.js","purposes.js","plugins/routing/client.js","plugins/routing/i18n.js","plugins/routing/routing.css","plugins/library/library.css","plugins/usage/i18n.js","quotaall.js","winone-routing-client.js","winone-ai-settings.css","winone-routing.js","winone-routing-bridge.js","winone-routing-shared.js","winone-routing-context.js","winone-routing.css"}{mux.Handle("GET /"+path,static)}
 mux.Handle("GET /icons/",static);mux.Handle("GET /fonts/",static)
 return mux
}
