package gateway

// HTTP Close does not close hijacked WebSockets. Explicitly disabling the
// external browser must disconnect its extension even while MCP hub stays on.
func(s *Server) WinOneExternalStopBrowser(){
 s.browser.mu.Lock()
 connections:=append([]*browserConn(nil),s.browser.conns...)
 s.browser.mu.Unlock()
 for _,connection:=range connections{connection.c.CloseNow()}
}
