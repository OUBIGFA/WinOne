package library

import (
 "encoding/json"
 "errors"
 "os"
 "path/filepath"
 "sync/atomic"
 "github.com/yetone/magpie/internal/agent"
)

type winOneExternalGateway struct{URL,Key string}
var winOneModelGateway atomic.Pointer[winOneExternalGateway]

// Installed reads this private status file without discovering client homes.
// Only the external worker mutates the actual Library and its client files.
func winOneExternalPluginServers()(map[string]PluginServer,bool){
 path:=os.Getenv("WINONE_AI_EXTERNAL_LIBRARY")
 if !filepath.IsAbs(path){return nil,false}
 data,err:=os.ReadFile(path);if err!=nil&&!errors.Is(err,os.ErrNotExist){return nil,true}
 var l Library;if len(data)>0&&json.Unmarshal(data,&l)!=nil{return nil,true}
 // Offer newly loaded tools before Library has been opened, but do not
 // persist or synchronize anything merely because Installed was rendered.
 l.adoptPlugins()
 out:=map[string]PluginServer{}
 for _,s:=range l.MCP{if s!=nil&&s.Plugin!=""{out[s.Plugin]=PluginServer{Name:s.Name,Plugin:s.Plugin,On:len(s.Agents)>0&&!s.Hidden}}}
 return out,true
}

// A user action may adopt cached tools; background plugin loading must never
// synchronize another application's files as a side effect.
func WinOneExternalAdoptTools()error{_,err:=adoptPluginTools();return err}

// Supplied over the Host-only channel. These calls retain the public AI
// gateway's authentication and model policy; the external listener serves no
// model route and never treats an MCP credential as a model permission.
func WinOneExternalUseGateway(url,key string){winOneModelGateway.Store(&winOneExternalGateway{url,key})}

func winOneExternalCanInstall(id string)error{
 if agent.WinOneExternalKey==""{return nil}
 if id!="magpie-image"&&id!="magpie-web-search"&&id!="magpie-vision"{return nil}
 endpoint:=winOneModelGateway.Load()
 if endpoint==nil||endpoint.URL==""||endpoint.Key==""{return errors.New("请先在 WinOne 本地网关中启用服务，再安装使用模型的内置 MCP")}
 return nil
}

// The browser command is backed by WinOne's separately authenticated local
// service. Keep the credential with the command's endpoint in client config.
func winOneExternalSelfCommand(s *Server){
 if agent.WinOneExternalKey==""{return}
 if s.Env==nil{s.Env=map[string]string{}}
 s.Env["WINONE_AI_MCP_KEY"]=agent.WinOneExternalKey
 if len(s.Args)>1&&s.Args[1]!="browser"{
  if endpoint:=winOneModelGateway.Load();endpoint!=nil{
   s.Env["MAGPIE_GATEWAY"],s.Env["WINONE_AI_MCP_KEY"]=endpoint.URL,endpoint.Key
  }
 }
}
