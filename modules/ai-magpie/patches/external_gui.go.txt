package gui

import (
 "errors"
 "net/http"
 "os/exec"
)

// These APIs reuse the upstream library and session handlers verbatim. There
// is no Magpie app shell and no warmStart in this independently owned worker.
func WinOneExternalHandler() http.Handler {
 warmOnce.Do(func(){})
 coldTurnsTaken=false
 mux:=http.NewServeMux()
 host:=winOneExternalWindows{}
 libraryRoutes(mux,host)
 sessionRoutes(mux,host)
 sessionManageRoutes(mux,host)
 return mux
}

type winOneExternalWindows struct{Windows}
func(winOneExternalWindows) OpenURL(value string)error{return exec.Command("rundll32.exe","url.dll,FileProtocolHandler",value).Start()}
func(winOneExternalWindows) OpenFolder(value string)error{return exec.Command("explorer.exe",value).Start()}
func(winOneExternalWindows) ChooseFolder(string)(string,error){return "",errors.New("choose the folder in the WinOne native picker")}
