 "context"
 "errors"

type winOneMediaModelsKey struct{}
// WinOne's gateway model list also limits its configured helper models.
// Ordinary upstream callers retain their original helper-model policy.
func WinOneMediaRequest(r *http.Request,models []string)*http.Request{
 who:=access.Caller(r.Context())
 // Public model selection and a named key's restrictions are both binding.
 // Resolve aliases and groups with the same rules as the actual request.
 if who.Restricted() {
  kept:=make([]string,0,len(models))
  for _,id:=range models {
   if g,ms,ok:=provider.FindGroup(id);ok {if groupAllowed(who,g,ms){kept=append(kept,id)};continue}
   if p,m,ok:=provider.Resolve(id);ok&&modelAllowed(who,p,m){kept=append(kept,id)}
  }
  models=kept
 }
 if len(models)==0 {models=[]string{"/"}}
 who.Models=models
 ctx:=context.WithValue(access.WithIdentity(r.Context(),who),winOneMediaModelsKey{},true)
 return r.WithContext(ctx)
}

// This entry is called only after authentication of the private Host bridge.
// Unknown/expired/off keys never fall back to upstream's loopback privileges.
func WinOneNamedRequest(w http.ResponseWriter,r *http.Request)(*http.Request,bool){
 secret:=r.Header.Get("X-WinOne-Gateway-Credential")
 remote:=r.Header.Get("X-WinOne-Remote-Addr")
 r.Header.Del("X-WinOne-Gateway-Credential");r.Header.Del("X-WinOne-Remote-Addr")
 if secret=="" {return r,true}
 r.RemoteAddr=remote
 if r.Method==http.MethodPost&&r.URL.Path==OIDCKeysPath {
  r.Header.Set("Authorization","Bearer "+secret);return r,true
 }
 who,err:=access.Check(secret)
 if err!=nil {writeError(w,provider.Chat,http.StatusUnauthorized,"gateway key is invalid, disabled or expired");return r,false}
 r=r.WithContext(access.WithIdentity(r.Context(),who))
 for _,name:=range []string{"X-API-Key","X-Goog-API-Key"}{r.Header.Del(name)}
 q:=r.URL.Query();q.Del("key");r.URL.RawQuery=q.Encode()
 return r,true
}
func winOneMediaRestricted(r *http.Request)bool{held,_:=r.Context().Value(winOneMediaModelsKey{}).(bool);return held}

// Check before vision's cache as well as before its next provider request.
// countHeld resolves original model/group IDs with Magpie's own rules.
func winOneMediaAllowed(r *http.Request,model string)error{
 if winOneMediaRestricted(r)&&countHeld(httptest.NewRecorder(),r,provider.Chat,model){return errors.New("model not allowed by WinOne gateway")}
 return nil
}

// Export only the classification used by restAfter. Error bodies never cross
// WinOne IPC; its host needs this bit to honor LimitRestOff across workers.
func WinOneLimitedResponse(status int,body []byte) bool {
 why:=failure(status,body)
 return status==http.StatusTooManyRequests||why==failQuota||why==failCredit||why==failRate
}

// Set before serving an owned provider bridge, never from public HTTP input.
// The outer gateway owns the user-request hooks and JS middleware.
var WinOneTransportOnly bool

// Fixed before serving: public account kinds carried by WinOne's otherwise
// ordinary HTTP bridges. This cannot select credentials or a native backend.
var WinOneMiddlewareAccounts map[string]string

// Only Host-owned bridges can inherit the source provider's Decisions support.
var WinOneDecisionProviders map[string]bool
func WinOneCanDecide(p provider.Provider) bool { return canDecide(p) }

// Only explicit API-key providers have a native transport. Account bridges
// must never be mistaken for API keys just because their local hop uses one.
var WinOneNativeProviders map[string]provider.Provider
func winOneNativeProvider(p provider.Provider) (provider.Provider,bool) {
 if actual,ok:=WinOneNativeProviders[p.ID];ok{return actual,true}
 if WinOneMiddlewareAccounts[p.ID]!="" {return p,false}
 return p,true
}
