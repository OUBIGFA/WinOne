cmake_minimum_required(VERSION 3.25)
project(WinOneNativeImage LANGUAGES CXX)
add_library(WinOne.NativeImage SHARED image.cpp)
target_compile_features(WinOne.NativeImage PRIVATE cxx_std_20)
target_compile_definitions(WinOne.NativeImage PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(WinOne.NativeImage PRIVATE ole32 windowscodecs gdi32 user32)
if(MSVC)
    target_compile_options(WinOne.NativeImage PRIVATE /utf-8 /W4 /WX /O2 /Gy)
    target_link_options(WinOne.NativeImage PRIVATE /OPT:REF /OPT:ICF)
endif()
