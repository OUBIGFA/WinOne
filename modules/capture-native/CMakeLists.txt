cmake_minimum_required(VERSION 3.25)
project(WinOneCaptureNative VERSION 1.4.11 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT WINONE_TOOLS3000_SOURCE)
    message(FATAL_ERROR "Pass the verified Tools3000 source directory")
endif()
set(UPSTREAM "${WINONE_TOOLS3000_SOURCE}")
if(NOT EXISTS "${UPSTREAM}/src/capture/CaptureOverlay.cpp")
    message(FATAL_ERROR "The complete upstream capture implementation is required")
endif()

find_package(spdlog CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(sol2 CONFIG REQUIRED)
find_package(Lua REQUIRED)
find_package(fmt CONFIG REQUIRED)
find_package(OpenCV CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
find_path(FFMPEG_INCLUDE_DIR libavcodec/avcodec.h REQUIRED)
find_library(AVCODEC_LIBRARY avcodec REQUIRED)
find_library(AVFORMAT_LIBRARY avformat REQUIRED)
find_library(AVUTIL_LIBRARY avutil REQUIRED)
find_library(SWSCALE_LIBRARY swscale REQUIRED)
find_library(SWRESAMPLE_LIBRARY swresample REQUIRED)
find_path(WEBP_INCLUDE_DIR webp/mux.h REQUIRED)
find_library(WEBP_LIBRARY NAMES libwebp webp REQUIRED)
find_library(WEBPMUX_LIBRARY NAMES libwebpmux webpmux REQUIRED)

# Build the actual pinned native implementation. The upstream top-level build
# also builds unrelated applications; this target selects only capture and the
# Core/OCR libraries its public entry points depend on.
file(GLOB_RECURSE CORE_SOURCES CONFIGURE_DEPENDS "${UPSTREAM}/src/core/*.cpp")
file(GLOB CAPTURE_SOURCES CONFIGURE_DEPENDS "${UPSTREAM}/src/capture/*.cpp")
file(GLOB OCR_SOURCES CONFIGURE_DEPENDS "${UPSTREAM}/src/ocr/*.cpp")
string(TIMESTAMP TOOLS3000_BUILD_DATE "%Y-%m-%d")
configure_file("${UPSTREAM}/src/core/Version.h.in" "${CMAKE_BINARY_DIR}/generated/Tools3000Version.h" @ONLY)

add_library(Tools3000Core SHARED ${CORE_SOURCES})
target_compile_definitions(Tools3000Core PRIVATE TOOLS3000CORE_EXPORTS)
target_include_directories(Tools3000Core PUBLIC "${UPSTREAM}/src" "${CMAKE_BINARY_DIR}/generated" "${LUA_INCLUDE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(Tools3000Core PUBLIC
    spdlog::spdlog nlohmann_json::nlohmann_json sol2::sol2 ${LUA_LIBRARIES} fmt::fmt ZLIB::ZLIB
    user32 shell32 advapi32 dbghelp psapi shlwapi winhttp crypt32 bcrypt uiautomationcore oleaut32 ole32
    dwmapi gdiplus imm32 shcore propsys windowsapp d2d1 dwrite d3d11 dxgi dcomp)

add_library(winone_capture_engine STATIC ${CAPTURE_SOURCES} ${OCR_SOURCES})
target_compile_definitions(winone_capture_engine PRIVATE PLUGIN_EXPORTS)
target_include_directories(winone_capture_engine PUBLIC "${UPSTREAM}/src" "${CMAKE_BINARY_DIR}/generated"
    "${FFMPEG_INCLUDE_DIR}" "${WEBP_INCLUDE_DIR}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(winone_capture_engine PUBLIC Tools3000Core ${OpenCV_LIBS}
    ${AVCODEC_LIBRARY} ${AVFORMAT_LIBRARY} ${AVUTIL_LIBRARY} ${SWSCALE_LIBRARY} ${SWRESAMPLE_LIBRARY}
    ${WEBP_LIBRARY} ${WEBPMUX_LIBRARY} d2d1 dwrite d3d11 dxgi dxguid dwmapi mfplat mfreadwrite mfuuid
    shlwapi shell32 user32 gdi32 ole32 oleaut32 advapi32 comctl32 windowscodecs winhttp windowsapp shcore imm32 ws2_32)
target_sources(winone_capture_engine PRIVATE toolbar_icons.cpp)
target_include_directories(winone_capture_engine PRIVATE "${UPSTREAM}/winone")

foreach(target Tools3000Core winone_capture_engine)
    # Keep upstream's Windows/COM include order and shared precompiled header.
    target_precompile_headers(${target} PRIVATE "${UPSTREAM}/src/core/CorePch.h")
    target_compile_definitions(${target} PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4 /bigobj /Zc:__cplusplus /permissive- /MP4 /Gy /Gw)
        target_link_options(${target} PRIVATE /OPT:REF /OPT:ICF)
    endif()
endforeach()

add_executable(WinOne.Capture WIN32 main.cpp capture_app.cpp capture_actions.cpp capture_recording.cpp capture_number.cpp host_bridge.cpp ocr_client.cpp draft_store.cpp recording_controller.cpp recording_options.cpp toolbar_options.cpp)
target_sources(WinOne.Capture PRIVATE "${UPSTREAM}/winone/winone_icons.rc")
target_include_directories(WinOne.Capture PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(WinOne.Capture PRIVATE winone_capture_engine)
if(MSVC)
    target_compile_options(WinOne.Capture PRIVATE /utf-8 /W4 /bigobj)
endif()
option(WINONE_BUILD_CAPTURE_PROBE "Build the same capture UI with generated private-desktop input" OFF)

        target_compile_options(WinOne.Capture.RecordingTests PRIVATE /utf-8 /W4)

    target_compile_definitions(WinOne.NativeCapture.Probe PRIVATE WINONE_CAPTURE_PRIVATE_PROBE)
    target_include_directories(WinOne.NativeCapture.Probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(WinOne.NativeCapture.Probe PRIVATE winone_capture_engine)
    if(MSVC)
